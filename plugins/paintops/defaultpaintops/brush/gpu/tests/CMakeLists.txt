cmake_minimum_required(VERSION 3.16)
project(KritaGpuBrushComputeTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(krita_gpu_brush_compute_test
    KisGpuBrushComputeTest.cpp
    ../KisGpuBrushCompositor.cpp)
target_link_libraries(krita_gpu_brush_compute_test PRIVATE Threads::Threads)
if(WIN32)
    target_sources(krita_gpu_brush_compute_test PRIVATE ../../../../../../krita/windows_gpu_preference.cpp)
    target_link_libraries(krita_gpu_brush_compute_test PRIVATE opengl32 gdi32 user32)
endif()
if(MSVC)
    target_compile_options(krita_gpu_brush_compute_test PRIVATE /W4 /fp:strict)
else()
    target_compile_options(krita_gpu_brush_compute_test PRIVATE -Wall -Wextra -ffp-contract=off)
endif()
enable_testing()
add_test(NAME GpuBrushCompute COMMAND krita_gpu_brush_compute_test)
